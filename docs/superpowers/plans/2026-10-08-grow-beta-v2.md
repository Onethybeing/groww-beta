**Goal:** Add the following to the live GROW Beta app, one reviewed PR each:

**Architecture:**
- **New logic goes in pure, tested modules:**
  - `lib/theme.ts`;
  - `lib/engine/next-step.ts`, `lib/engine/xp.ts` and `lib/engine/quests.ts`;
  - `lib/i18n/*`;
  - `lib/insights.ts`.
- **UI preferences** (theme, language, anonymous id) live in a new small persisted store, `usePrefs` (localStorage
  key `grow-ui`), kept separate from the app data store `useApp` (key `groww-genz`).
- **Colours** become semantic CSS variables that switch under `.dark`.
- **Analytics** is a write-only, allow-listed `/api/events` route backed by Upstash Redis, which does nothing when
  the Redis environment variables are missing.

**Tech Stack:** Next.js 16 App Router, Tailwind v4, Zustand 5, Zod, lightweight-charts 5, `@axe-core/playwright`,
Lighthouse CLI, `@upstash/redis`, ffmpeg (installed at
`…/WinGet/Packages/Gyan.FFmpeg…/bin/ffmpeg`).

**Spec:** `docs/superpowers/specs/2026-10-08-grow-beta-v2-design.md` (extends `2026-10-07-groww-genz-starter-design.md`, revision 3).

## Review Focus

1. **Theme flash or hydration mismatch.** A first paint in dark mode must not flash light, and React must not warn
   about a mismatched `class` on `<html>`. Pinned in Task 1 by the e2e step "reload in dark → no light flash class
   and no hydration warning".
2. **Long Hindi strings at 360px.** Hindi labels must not overflow buttons, tabs or the tab bar. Pinned in Task 7 by
   the overflow crawl in Hindi at 360px.
3. **Service worker serving stale HTML after a deploy.** A new deploy must replace the cached shell. Pinned in Task 4:
   the cache name includes the build id and old caches are deleted on `activate`; a test asserts that.
4. **Bad analytics payloads.** Unknown events, missing ids, oversized bodies and floods return 400 or 429 and never
   write to Redis. Pinned in Task 8 by route tests.
5. **Quest week boundaries under demo time travel.** Quests reset on the ISO week of the *demo* date, and quests from
   a previous week aren't double-counted. Pinned in Task 6 by unit tests across a week boundary.

```ts
export type ThemePref = "system" | "light" | "dark";
export const resolveTheme = (pref: ThemePref, prefersDark: boolean): "light" | "dark" =>
  pref === "system" ? (prefersDark ? "dark" : "light") : pref;

/** Runs in <head> before paint: applies .dark from the persisted pref to avoid a light flash. */
export const THEME_BOOT_SCRIPT = `(function(){try{var p=JSON.parse(localStorage.getItem("grow-ui")||"{}").state||{};var t=p.theme||"system";var d=t==="dark"||(t==="system"&&matchMedia("(prefers-color-scheme: dark)").matches);document.documentElement.classList.toggle("dark",d);document.documentElement.style.colorScheme=d?"dark":"light";}catch(e){}})();`;
```

- [ ] **Step 3: Implement `lib/store/prefs.ts`**

```ts
"use client";
import { create } from "zustand";
import { persist, createJSONStorage } from "zustand/middleware";
import { useEffect, useSyncExternalStore } from "react";
import { resolveTheme, type ThemePref } from "@/lib/theme";

export type Lang = "en" | "hi";
interface Prefs { theme: ThemePref; lang: Lang; anonId: string; setTheme(t: ThemePref): void; setLang(l: Lang): void }
const newId = () => (typeof crypto !== "undefined" && "randomUUID" in crypto ? crypto.randomUUID() : `a-${Date.now()}`);

export const usePrefs = create<Prefs>()(persist((set) => ({
  theme: "system", lang: "en", anonId: newId(),
  setTheme: (theme) => set({ theme }),
  setLang: (lang) => set({ lang }),
}), { name: "grow-ui", version: 1, storage: createJSONStorage(() => localStorage) }));

const mq = () => (typeof window === "undefined" ? null : window.matchMedia("(prefers-color-scheme: dark)"));
export function useResolvedTheme(): "light" | "dark" {
  const pref = usePrefs((s) => s.theme);
  const prefersDark = useSyncExternalStore(
    (cb) => { const m = mq(); m?.addEventListener("change", cb); return () => m?.removeEventListener("change", cb); },
    () => mq()?.matches ?? false,
    () => false,
  );
  const resolved = resolveTheme(pref, prefersDark);
  useEffect(() => {
    document.documentElement.classList.toggle("dark", resolved === "dark");
    document.documentElement.style.colorScheme = resolved;
  }, [resolved]);
  return resolved;
}
```

```ts
import { personaFor, type Answers } from "./persona";
import { computeStreak } from "./streak";
import { nextSipDate, type Sip, type SipInstalment } from "./sips";
import { referralFreezes } from "./referrals";
import { fundForCategory } from "@/lib/market/instruments";
import type { Txn } from "./portfolio";

export type NextStepId = "welcome" | "lesson" | "practice" | "start_sip" | "resume_sip" | "keep_streak" | "next_lesson" | "invite" | "quest";
export interface NextStepInput {
  answers: Answers | null; welcomeSeen: boolean; lessons: Record<string, unknown>; transactions: Txn[]; timeMachineRuns: number;
  sips: Sip[]; instalments: SipInstalment[]; referrals: unknown[]; referredBy: string | null;
}
export interface NextStep { id: NextStepId; href: string; vars: Record<string, string | number> }
const LESSON_ORDER = ["mutual-funds-sip", "index-funds", "ups-and-downs"];

export function nextBestStep(s: NextStepInput, today: string): NextStep {
  if (!s.answers && !s.welcomeSeen) return { id: "welcome", href: "/?welcome=1", vars: {} };
  const persona = s.answers ? personaFor(s.answers) : null;
  const fund = fundForCategory(persona?.suggestedCategory ?? "index");
  const lessonsDone = Object.keys(s.lessons);
  if (lessonsDone.length === 0) {
    const first = persona?.id === "curious-explorer" ? "index-funds" : persona?.id === "goal-saver" ? "ups-and-downs" : "mutual-funds-sip";
    return { id: "lesson", href: `/learn/${first}`, vars: {} };
  }
  if (s.transactions.length === 0 && s.timeMachineRuns === 0) return { id: "practice", href: `/funds/${fund.symbol}`, vars: { fund: fund.short } };
  const live = s.sips.filter((x) => x.status !== "cancelled");
  const active = live.filter((x) => x.status === "active");
  if (live.length === 0) return { id: "start_sip", href: `/funds/${fund.symbol}/sip`, vars: { amount: persona?.suggestedAmount ?? 100, fund: fund.short } };
  if (active.length === 0) return { id: "resume_sip", href: `/sips/${live[0].id}`, vars: {} };
  const freezes = 1 + referralFreezes(s.referrals.length, s.referredBy !== null);
  const streak = computeStreak(s.instalments.map((i) => i.date), today, freezes).current;
  if (streak < 3) {
    const next = active.map((x) => nextSipDate(x, s.instalments, today)).sort()[0];
    return { id: "keep_streak", href: `/sips/${active[0].id}`, vars: { date: next, months: streak } };
  }
  const nextLesson = LESSON_ORDER.find((id) => !lessonsDone.includes(id));
  if (nextLesson) return { id: "next_lesson", href: `/learn/${nextLesson}`, vars: {} };
  if (s.referrals.length === 0) return { id: "invite", href: "/refer", vars: {} };
  return { id: "quest", href: "/quests", vars: {} };
}
```

```ts
export function isoWeekKey(date: string): string {
  const d = new Date(`${date}T00:00:00Z`);
  const day = (d.getUTCDay() + 6) % 7; // Mon=0
  d.setUTCDate(d.getUTCDate() - day + 3); // Thursday of this week
  const year = d.getUTCFullYear();
  const jan4 = new Date(Date.UTC(year, 0, 4));
  const week = 1 + Math.round(((d.getTime() - jan4.getTime()) / 86_400_000 - 3 + ((jan4.getUTCDay() + 6) % 7)) / 7);
  return `${year}-W${String(week).padStart(2, "0")}`;
}
```

```ts
    firstTrade: i.transactions.length > 0 ? 10 : 0,

    streak: Math.max(0, i.streakMonths - 1) * 15,
    notes: Math.min(5, i.reflections.length) * 5,
    quests: i.questsCompleted * 20,
  };
  return { total: Object.values(breakdown).reduce((a, b) => a + b, 0), breakdown };
}

export function levelFor(xp: number) {
  let idx = 0;
  LEVELS.forEach((l, k) => { if (xp >= l.min) idx = k; });
  const cur = LEVELS[idx]; const next = LEVELS[idx + 1];
  return { name: cur.name, min: cur.min, next: next ? next.min : null, progress: next ? (xp - cur.min) / (next.min - cur.min) : 1 };
```

```ts
import { isoWeekKey, monthKey, addDays } from "./dates";
import { nextSipDate, type Sip, type SipInstalment } from "./sips";
import type { GameEvent } from "./xp";

export type QuestId = "explainers_2" | "lesson_1" | "time_machine_1" | "note_1" | "sip_on_track";
export interface QuestCtx {
  events: GameEvent[]; lessons: Record<string, { completedAt: string }>; reflections: { date: string }[];
  sips: Sip[]; instalments: SipInstalment[]; today: string;
}
const TARGET: Record<QuestId, number> = { explainers_2: 2, lesson_1: 1, time_machine_1: 1, note_1: 1, sip_on_track: 1 };

export function weeklyQuests(week: string): QuestId[] {
  const n = Number(week.slice(-2));
  return n % 2 === 0 ? ["explainers_2", "lesson_1", "sip_on_track"] : ["time_machine_1", "note_1", "sip_on_track"];
}

const inWeek = (date: string, week: string) => isoWeekKey(date) === week;

export function questProgress(id: QuestId, c: QuestCtx, week: string) {
  let done = 0;
  if (id === "explainers_2") done = c.events.filter((e) => e.type === "explainer_open" && inWeek(e.date, week)).length;
  if (id === "time_machine_1") done = c.events.filter((e) => e.type === "time_machine" && inWeek(e.date, week)).length;
  if (id === "lesson_1") done = Object.values(c.lessons).filter((l) => inWeek(l.completedAt, week)).length;
  if (id === "note_1") done = c.reflections.filter((r) => inWeek(r.date, week)).length;
  if (id === "sip_on_track") {
    const paid = c.instalments.some((i) => monthKey(i.date) === monthKey(c.today));
    const upcoming = c.sips.some((s) => s.status === "active" && monthKey(nextSipDate(s, c.instalments, c.today)) === monthKey(c.today));
    done = paid || upcoming ? 1 : 0;
  }
  const target = TARGET[id];
  return { done: Math.min(done, target), target, complete: done >= target };
}

/** Count of quests completed in every week from `fromWeek` to `toWeek`, walking week by week (Mondays). */
export function completedQuestCount(c: QuestCtx, fromWeek: string, toWeek: string): number {
  let total = 0;
  let cursor = c.today;
  while (isoWeekKey(cursor) > toWeek) cursor = addDays(cursor, -7);
  const seen = new Set<string>();
  for (let i = 0; i < 520; i++) {
    const wk = isoWeekKey(cursor);
    if (wk < fromWeek) break;
    if (!seen.has(wk)) {
      seen.add(wk);
      const asOf = { ...c, today: cursor };
      total += weeklyQuests(wk).filter((q) => questProgress(q, q === "sip_on_track" ? asOf : c, wk).complete).length;
    }
    cursor = addDays(cursor, -7);
  }
  return total;
}

export function monthlyChallenge(c: QuestCtx, suggested: number) {
  const invested = c.instalments.filter((i) => monthKey(i.date) === monthKey(c.today)).reduce((a, i) => a + i.amount, 0);
  const target = suggested * 2;
  return { target, invested, complete: invested >= target };
}
```
